### Step 2：嵌入服务 + 向量检索

**新建** `backend/app/rag/embedding.py`：

核心功能：
- `get_embedding_client() -> EmbeddingClient` — 嵌入客户端工厂（和 LLM 工厂类似）
- `EmbeddingClient.embed(text: str) -> list[float]` — 调用嵌入 API 生成向量
- 无 API key 时返回随机向量（MVP 兜底，保证不报错）
- 支持的 provider：openai（text-embedding-3-small）、glm、deepseek

```python
@dataclass(frozen=True)
class EmbeddingClient:
    provider: str
    api_key: str
    base_url: str
    model: str

    async def embed(self, text: str) -> list[float]:
        if not self.api_key:
            # 兜底：返回随机向量，长度 1536
            import hashlib, struct
            seed = hashlib.sha256(text.encode()).digest()
            return [struct.unpack('f', seed[i:i+4])[0] / 1e10 for i in range(0, min(len(seed)*4, 1536*4), 4)][:1536]

        async with httpx.AsyncClient(timeout=30) as client:
            response = await client.post(
                f"{self.base_url.rstrip('/')}/embeddings",
                headers={"Authorization": f"Bearer {self.api_key}"},
                json={"model": self.model, "input": text[:8000]},
            )
            response.raise_for_status()
            return response.json()["data"][0]["embedding"]
```

```python
async def chat_stream(db, user_id, conv_id, query):
    # 1. 保存用户消息
    user_msg = Message(conversation_id=conv_id, role="user", content=query)
    db.add(user_msg)
    await db.flush()

    # 2. RAG 检索
    contexts = await retrieval.search_similar(db, user_id, query, top_k=5)
    rag_messages = prompts.build_rag_messages(query, contexts)

    # 3. 流式调用 LLM
    full_answer = ""
    async for chunk in llm.chat_stream(rag_messages):
        full_answer += chunk
        yield chunk  # SSE 推送给前端

    # 4. 保存 AI 消息（含来源）
    ai_msg = Message(
        conversation_id=conv_id,
        role="assistant",
        content=full_answer,
        sources=[{"note_id": str(c["note_id"]), "title": c["note_title"]} for c in contexts],
    )
    db.add(ai_msg)
    await db.flush()
```

**修改** `backend/app/rag/llm.py`：新增 `chat_stream` 方法（SSE 流式）

```python
async def chat_stream(self, messages: list[dict[str, str]]) -> AsyncGenerator[str, None]:
    """流式对话，yield 每个 chunk 的 content delta。"""
    async with httpx.AsyncClient(timeout=60) as client:
        async with client.stream(
            "POST",
            f"{self.base_url.rstrip('/')}/chat/completions",
            headers={"Authorization": f"Bearer {self.api_key}"},
            json={"model": self.model, "messages": messages, "temperature": 0.3, "stream": True},
        ) as response:
            response.raise_for_status()
            async for line in response.aiter_lines():
                if line.startswith("data: "):
                    data = line[6:]
                    if data == "[DONE]":
                        break
                    chunk = json.loads(data)
                    delta = chunk.get("choices", [{}])[0].get("delta", {})
                    if content := delta.get("content"):
                        yield content
```

```python
async def _embed_note(db: AsyncSession, note_id: uuid.UUID, text: str):
    """将笔记文本分块并嵌入。短文本（<500 字）1 块，长文本按段落分块。"""
    chunks = _split_text(text, max_chunk_size=500)
    embedding_client = get_embedding_client()

    for index, chunk_text in enumerate(chunks):
        content_hash = hashlib.sha256(chunk_text.encode()).hexdigest()
        vector = await embedding_client.embed(chunk_text)
        chunk = NoteChunk(
            note_id=note_id,
            chunk_index=index,
            chunk_text=chunk_text,
```

分块逻辑：
```python
def _split_text(text: str, max_chunk_size: int = 500) -> list[str]:
    if len(text) <= max_chunk_size:
        return [text]
    paragraphs = text.split("\n\n")
    chunks, current = [], ""
    for p in paragraphs:
        if len(current) + len(p) > max_chunk_size and current:
            chunks.append(current.strip())
            current = p
        else:
            current = current + "\n\n" + p if current else p
    if current.strip():
        chunks.append(current.strip())
    return chunks
```

```python
async def start_interview(db, user_id, scope, num_questions=5):
    # 1. 根据范围获取知识点
    if scope == "weak_points":
        notes = await _get_weak_notes(db, user_id)  # mastery_level < 2 或复习评分低的
    else:
        notes = await _get_all_notes(db, user_id)

    # 2. 从知识点中提取 topics
    topics = [f"{note.title}: {note.content[:100]}" for note in notes[:20]]

    # 3. AI 生成面试题

    # 4. 创建 session + questions

    db.add(session)
    await db.flush()

    for i, q in enumerate(questions):
        question = InterviewQuestion(
            session_id=session.id,
            note_id=notes[i].id if i < len(notes) else None,
            question=q["question"],
            question_order=i + 1,
        )
        db.add(question)

    await db.flush()
    await db.refresh(session)
    return session
```

```python
from sse_starlette.sse import EventSourceResponse

@router.post("/conversations/{conv_id}/chat")
async def chat(conv_id: uuid.UUID, data: ChatRequest, ...):
    async def event_generator():
        async for chunk in chat_service.chat_stream(db, current_user.id, conv_id, data.query):
            yield {"data": json.dumps({"content": chunk}, ensure_ascii=False)}
        # 发送来源信息
        yield {"data": json.dumps({"done": True, "sources": [...]}, ensure_ascii=False)}
    return EventSourceResponse(event_generator())
```

### 前端 SSE 读取

```typescript
const response = await fetch('/api/v1/chat/conversations/xxx/chat', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
  body: JSON.stringify({ query }),
})
const reader = response.body!.getReader()
const decoder = new TextDecoder()
while (true) {
  const { done, value } = await reader.read()
  if (done) break
  const text = decoder.decode(value)
  // 解析 SSE: data: {"content": "..."}\n\n
  const lines = text.split('\n').filter(l => l.startsWith('data: '))
  for (const line of lines) {
    const data = JSON.parse(line.slice(6))
    if (data.done) { /* 完成 */ }
    else { onChunk(data.content) }
  }
}
```

### pgvector 向量存储和检索

```python
import struct

def vector_to_bytes(vec: list[float]) -> bytes:
    """将 float 列表转为 pgvector 存储格式。"""
    return struct.pack(f'{len(vec)}f', *vec)

def bytes_to_vector(data: bytes) -> list[float]:
    """从 bytes 还原 float 列表。"""
    return list(struct.unpack(f'{len(data)//4}f', data))
```
