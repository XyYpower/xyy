### Step 2：AI 知识点提取服务

**新建** `backend/app/rag/extractor.py`：

核心函数：
- `extract_knowledge_points(text: str) -> list[dict]` — 调用 LLM 从文本中提取知识点
- 输入：一段技术文章/面经/代码
- 输出：`[{title, content}, ...]` 结构化知识点列表
- Prompt 要求 LLM 返回 JSON：`{"items": [{"title": "...", "content": "..."}]}`
- 没有配置 API key 时，用简单的规则提取（按段落拆分，取前 N 段作为知识点）

```python
async def extract_knowledge_points(text: str) -> list[dict[str, str]]:
    """从文本中提取知识点。"""
    llm = get_llm()
    if not llm.api_key:
        return _fallback_extract(text)

    prompt = (
        "你是编程知识点提取器。从用户给出的技术文章/面经/代码中，提取出独立可学习的知识点。"
        "每个知识点包含 title（简短标题）和 content（详细说明）。"
        "只返回 JSON：{\"items\":[{\"title\":\"...\",\"content\":\"...\"}]}"
    )
    try:
        raw = await llm.chat_json([
            {"role": "system", "content": prompt},
            {"role": "user", "content": text[:4000]},  # 限制输入长度
        ])
        items = json.loads(raw).get("items", [])
        return [item for item in items if item.get("title") and item.get("content")]
    except Exception:
        return _fallback_extract(text)


def _fallback_extract(text: str) -> list[dict[str, str]]:
    """无 API key 时按段落拆分。"""
    paragraphs = [p.strip() for p in text.split("\n\n") if len(p.strip()) > 20]
    return [
        {"title": p[:50] + ("..." if len(p) > 50 else ""), "content": p}
        for p in paragraphs[:10]  # 最多 10 个
    ]
```

`confirm_import` 的关键逻辑：
```python
async def confirm_import(db, user_id, job_id):
    job = await db.get(ImportJob, job_id)
    # 验证 job 属于当前用户且状态为 draft
    drafts = await get_selected_drafts(db, job_id)
    notes = []
    for draft in drafts:
        note = Note(
            user_id=user_id,
            title=draft.title,
            content=draft.content,
            source_type="imported",
            source_url=job.source_url,
        )
        db.add(note)
        await db.flush()
        await db.refresh(note)
        draft.note_id = note.id  # 回填 note_id
        notes.append(note)
        # 自动生成复习卡片（复用 V2.1 的逻辑）
        await review_service.generate_cards_for_note(db, user_id, note.id)
    job.status = "confirmed"
    await db.flush()
    return notes
```

## 技术要点

### URL 抓取（简单实现）

```python
import httpx
import re

async def fetch_url_text(url: str) -> str:
    """抓取 URL 并提取纯文本。"""
    async with httpx.AsyncClient(timeout=15, follow_redirects=True) as client:
        response = await client.get(url, headers={"User-Agent": "KnowBase/1.0"})
        response.raise_for_status()
    html = response.text
    # 去除 script/style 标签
    html = re.sub(r"<(script|style)[^>]*>.*?</\1>", "", html, flags=re.DOTALL | re.IGNORECASE)
    # 去除所有 HTML 标签
    text = re.sub(r"<[^>]+>", " ", html)
    # 合并空白
    text = re.sub(r"\s+", " ", text).strip()
    return text[:8000]  # 限制长度
```
